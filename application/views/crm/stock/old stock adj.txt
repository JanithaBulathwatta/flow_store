(function() {
    'use strict';

    var MASTER_BATCHES = {};
    var ADJ_NO = <?php echo json_encode($adjustment_no); ?>;
    var currentPage = 1;
    var pageSize = 50;
    var currentFilters = {
        search: '',
        warehouse: '',
        category: '',
        brand: '',
        show: 'all'
    };

    document.getElementById('adjNoDisplay').textContent = ADJ_NO;

    function formatCurrency(n) {
        if(n == null || isNaN(n)) return '—';
        var a = Math.abs(n);
        return (n < 0 ? '-' : '') + 'LKR ' + a.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    }

    function formatDate(d) {
        if(!d) return '—';
        var x = new Date(d);
        return isNaN(x) ? d : x.toLocaleDateString('en-GB', {day:'2-digit', month:'short', year:'numeric'});
    }

    function showToast(msg, type) {
        var colors = {success:'#22c55e', danger:'#ef4444', warning:'#f59e0b', primary:'#405189'};
        var b = document.getElementById('__tb');
        if(!b) { b = document.createElement('div'); b.id = '__tb'; document.body.appendChild(b); }
        var t = document.createElement('div');
        t.className = 't-itm';
        t.style.background = colors[type] || colors.primary;
        t.innerHTML = '<i class="ri-checkbox-circle-line me-2"></i>' + msg;
        b.appendChild(t);
        requestAnimationFrame(function() { t.style.opacity = 1; t.style.transform = 'translateY(0)'; });
        setTimeout(function() { t.style.opacity = 0; setTimeout(function() { t.remove(); }, 300); }, 3200);
    }

    function getVariance(product) {
        if(product.counted === null || product.counted === '') return null;
        return parseFloat(product.counted) - parseFloat(product.systemQty);
    }

    function updateMainProgressAndCards() {
        var totalProducts = 0;
        var countedProducts = 0;
        var positiveVariance = 0;
        var negativeVariance = 0;
        var totalSystemValue = 0;
        
        for(var batchNo in MASTER_BATCHES) {
            var batch = MASTER_BATCHES[batchNo];
            if(batch.products) {
                batch.products.forEach(function(product) {
                    totalProducts++;
                    totalSystemValue += product.systemQty * product.avg_cost;
                    
                    if(product.counted !== null && product.counted !== '') {
                        countedProducts++;
                        var v = getVariance(product);
                        if(v > 0) positiveVariance++;
                        if(v < 0) negativeVariance++;
                    }
                });
            }
        }
        
        var percentage = totalProducts ? Math.round((countedProducts / totalProducts) * 100) : 0;
        
        console.log('=== PROGRESS UPDATE ===');
        console.log('Total:', totalProducts, 'Counted:', countedProducts, 'Percentage:', percentage + '%');
        
        var progressBar = document.getElementById('progressBar');
        if(progressBar) progressBar.style.width = percentage + '%';
        
        var progressText = document.getElementById('progressText');
        if(progressText) progressText.textContent = countedProducts + ' / ' + totalProducts + ' counted';
        
        var posSpan = document.getElementById('posVarCount');
        if(posSpan) posSpan.textContent = '+' + positiveVariance + ' pos';
        var negSpan = document.getElementById('negVarCount');
        if(negSpan) negSpan.textContent = '−' + negativeVariance + ' neg';
        
        var cards = [
            {bg: '#eff6ff', ic: '#1d4ed8', icon: 'ri-list-check-3', label: 'Total Lines', value: totalProducts, sub: countedProducts + ' counted, ' + (totalProducts - countedProducts) + ' pending'},
            {bg: '#fef9c3', ic: '#854d0e', icon: 'ri-loader-4-line', label: 'Count Progress', value: percentage + '%', sub: countedProducts + ' / ' + totalProducts + ' lines'},
            {bg: '#dcfce7', ic: '#166534', icon: 'ri-arrow-up-circle-line', label: 'Positive Variance', value: positiveVariance + ' lines', sub: 'Stock higher than system'},
            {bg: '#fee2e2', ic: '#991b1b', icon: 'ri-arrow-down-circle-line', label: 'Negative Variance', value: negativeVariance + ' lines', sub: 'Stock lower than system'},
            {bg: '#fdf4ff', ic: '#7e22ce', icon: 'ri-money-dollar-circle-line', label: 'System Stock Value', value: formatCurrency(totalSystemValue), sub: 'At avg cost'}
        ];
        
        var kpiHtml = cards.map(function(c) {
            return '<div class="col-sm-6 col-xl"><div class="card custom-card"><div class="card-body kc" style="background:' + c.bg + ';border-color:' + c.bg + ';">' +
                '<div class="ki" style="background:rgba(255,255,255,.5);"><i class="' + c.icon + '" style="color:' + c.ic + ';font-size:1.15rem;"></i></div>' +
                '<div><div class="kv" style="color:' + c.ic + ';">' + c.value + '</div><div class="kl">' + c.label + '</div><div class="ks">' + c.sub + '</div></div>' +
                '<div class="kbg" style="background:' + c.ic + ';"></div>' +
                '</div></div></div>';
        }).join('');
        
        var kpiRow = document.getElementById('kpiRow');
        if(kpiRow) kpiRow.innerHTML = kpiHtml;
    }

    function loadData() {
        var url = '<?php echo base_url("StockAdjustment/get_stock_data_grouped"); ?>';
        var params = [];
        
        if(currentFilters.warehouse) params.push('warehouse_id=' + currentFilters.warehouse);
        if(currentFilters.category) params.push('category_id=' + currentFilters.category);
        if(currentFilters.brand) params.push('brand_id=' + currentFilters.brand);
        if(currentFilters.search) params.push('search=' + encodeURIComponent(currentFilters.search));
        
        if(params.length) url += '?' + params.join('&');

        fetch(url)
            .then(response => response.json())
            .then(data => {
                MASTER_BATCHES = data.batches || {};
                
                // DEBUG: Print all product IDs
                console.log('=== ALL PRODUCT IDs IN MASTER_BATCHES ===');
                var allIds = [];
                for(var b in MASTER_BATCHES) {
                    if(MASTER_BATCHES[b].products) {
                        MASTER_BATCHES[b].products.forEach(function(p) {
                            allIds.push(p.id);
                            console.log('Product ID:', p.id, 'Name:', p.product_name);
                        });
                    }
                }
                console.log('Total products:', allIds.length);
                console.log('All IDs:', allIds.sort(function(a,b){return a-b}));
                
                for(var b in MASTER_BATCHES) {
                    if(MASTER_BATCHES[b].products) {
                        MASTER_BATCHES[b].products.forEach(function(p) { 
                            p.counted = null; 
                        });
                    }
                }
                
                renderTable();

    function getSelectText(selectId, value) {
        var select = document.getElementById(selectId);
        for(var i = 0; i < select.options.length; i++) {

    function getFilteredProductCount() {
        var count = 0;
        for(var batchNo in MASTER_BATCHES) {
            var batch = MASTER_BATCHES[batchNo];
            if(!shouldShowBatch(batch)) continue;
            count += getFilteredProductsInBatch(batch).length;
        }
        return count;
    }
    
    function shouldShowBatch(batch) {
        if(currentFilters.warehouse && batch.warehouse_id != currentFilters.warehouse) return false;
        return true;
    }
    
    function getFilteredProductsInBatch(batch) {
        return batch.products.filter(function(product) {
            if(currentFilters.search && !(product.product_name.toLowerCase().includes(currentFilters.search) || 
                (product.sku || '').toLowerCase().includes(currentFilters.search) || 
                (batch.batch_no || '').toLowerCase().includes(currentFilters.search))) return false;
            
            if(currentFilters.category && product.category_id != currentFilters.category) return false;
            if(currentFilters.brand && product.brand_id != currentFilters.brand) return false;
            
            var v = getVariance(product);
            if(currentFilters.show === 'counted' && (!product.counted || product.counted === '')) return false;
            if(currentFilters.show === 'uncounted' && product.counted && product.counted !== '') return false;
            if(currentFilters.show === 'variance' && (v === null || v === 0)) return false;
            if(currentFilters.show === 'pos' && (v === null || v <= 0)) return false;
            if(currentFilters.show === 'neg' && (v === null || v >= 0)) return false;
            
            return true;
        });
    }

    window.handleCountInput = function(inputElement) {
    console.log('=== handleCountInput CALLED ===');
    var productId = parseInt(inputElement.getAttribute('data-id'));
    var newValue = inputElement.value.trim();
    var systemQty = parseFloat(inputElement.getAttribute('data-system-qty')) || 0;
    var avgCost = parseFloat(inputElement.getAttribute('data-avg-cost')) || 0;
    var countedValue = (newValue === '') ? null : parseFloat(newValue);
    
    console.log('Product ID:', productId, 'New Value:', countedValue);
    
    var found = false;
    
    for(var batchNo in MASTER_BATCHES) {
        var batch = MASTER_BATCHES[batchNo];
        if(batch.products) {
            for(var i = 0; i < batch.products.length; i++) {
                var product = batch.products[i];
                if(product.id == productId) {
                    product.counted = countedValue;
                    found = true;
                    console.log('✅ UPDATED! Product:', product.product_name, 'Counted:', countedValue);
                    break;
                }
            }
            if(found) break;
        }
    }
    
    if(!found) {
        console.error('❌ Product NOT FOUND! ID:', productId);
        return;
    }
    
    var v = (countedValue !== null) ? countedValue - systemQty : null;
    
    // Update input styling
    inputElement.className = 'new-qty-input';
    if(countedValue !== null) {
        if(v > 0) inputElement.className += ' pos-var';
        else if(v < 0) inputElement.className += ' neg-var';
        else inputElement.className += ' zero-var';
    }
    
    var row = inputElement.closest('tr');
    if(row) {
        // Update row class
        if(countedValue !== null) {
            if(v > 0) row.className = 'variance-pos';
            else if(v < 0) row.className = 'variance-neg';
            else row.className = 'variance-zero dirty';
        } else {
            row.className = '';
        }
        
        // Get all cells
        var cells = row.querySelectorAll('td');
        
        // Column indices (0-based):
        // 0: checkbox
        // 1: product name/SKU
        // 2: category/brand
        // 3: warehouse
        // 4: batch
        // 5: expiry
        // 6: system qty (text-center)
        // 7: counted qty (input field - this is where we are)
        // 8: unit cost column  
        // 9: variance/value impact column  
        // 10: status column
        
        // Update variance column (index 8)
        // Update combined variance/value impact column (index 8)
if(cells[8]) {
    cells[8].innerHTML = '<span class="text-muted fs-12">' + formatCurrency(avgCost) + '</span>';
}

// Update combined variance/value impact column (index 9) - NOW SECOND
if(cells[9]) {
    if(v === null) {
        cells[9].innerHTML = '<span class="text-muted">—</span>';
    } else if(v === 0) {
        cells[9].innerHTML = '<div><span class="var-zero">0</span></div><div class="fs-11 text-muted mt-1">No impact</div>';
    } else {
        var impactValue = v * avgCost;
        var sign = v > 0 ? '+' : '';
        var arrow = v > 0 ? '↑' : '↓';
        var colorClass = v > 0 ? 'text-success' : 'text-danger';
        cells[9].innerHTML = '<div><span class="' + (v > 0 ? 'var-pos' : 'var-neg') + '">' + sign + v + ' units</span></div>' +
                           '<div class="fs-11 ' + colorClass + ' mt-1">' + arrow + ' ' + formatCurrency(Math.abs(impactValue)) + '</div>';
    }
}

// Update status column (index 10) - NOT 11 anymore!
if(cells[10]) {
    var status = (countedValue === null) ? 'uncounted' : (v === 0 ? 'match' : (v > 0 ? 'pos' : 'neg'));
    var statusHtml = {
        uncounted: '<span class="sc-chip" style="background:#f1f5f9;color:#64748b;"><i class="ri-time-line me-1"></i>Pending</span>',
        match: '<span class="sc-chip scc-ok"><i class="ri-checkbox-circle-fill me-1"></i>Match</span>',

}

    function renderPagination(total) {
        var pg = document.getElementById('pagination');
        if(!pg) return;
        var size = pageSize === 'all' ? total : pageSize;
        var pages = Math.ceil(total / size);
        if(pages <= 1) { pg.innerHTML = ''; return; }
        var h = '<li class="page-item ' + (currentPage === 1 ? 'disabled' : '') + '"><a class="page-link" href="#" onclick="goToPage(' + (currentPage-1) + ');return false;">&laquo;</a></li>';
        for(var i = 1; i <= pages; i++) {
            if(i === 1 || i === pages || Math.abs(i - currentPage) <= 1)
                h += '<li class="page-item ' + (i === currentPage ? 'active' : '') + '"><a class="page-link" href="#" onclick="goToPage(' + i + ');return false;">' + i + '</a></li>';
            else if(Math.abs(i - currentPage) === 2)
                h += '<li class="page-item disabled"><a class="page-link">…</a></li>';
        }
        h += '<li class="page-item ' + (currentPage === pages ? 'disabled' : '') + '"><a class="page-link" href="#" onclick="goToPage(' + (currentPage+1) + ');return false;">&raquo;</a></li>';
        pg.innerHTML = h;
    }

    window.openApproveModal = function() {
    var withVar = [];
    var totalImpact = 0;
    var countedTotal = 0;
    var totalSystemQty = 0;
    var totalCountedQty = 0;
    
    for(var batchNo in MASTER_BATCHES) {
        var batch = MASTER_BATCHES[batchNo];
        batch.products.forEach(function(p) {
            // Only include products that have been counted
            if(p.counted !== null && p.counted !== '') {
                countedTotal++;
                totalCountedQty += parseFloat(p.counted);
                totalSystemQty += p.systemQty;
                
                var v = parseFloat(p.counted) - p.systemQty;
                if(v !== 0) {
                    withVar.push(p);

    function updateStatusUI(status) {
        var chips = { draft: '<i class="ri-edit-line me-1"></i>Draft', submitted: '<i class="ri-send-plane-line me-1"></i>Submitted', approved: '<i class="ri-checkbox-circle-fill me-1"></i>Approved' };
        var el = document.getElementById('statusChip');
        if(el) el.innerHTML = chips[status] || chips.draft;
        
        var steps = ['Open', 'Count', 'Review', 'Submit', 'Approve', 'Posted'];
        var idx = {draft: 1, submitted: 3, approved: 5};
        var activeIdx = idx[status] || 1;
        var wf = document.getElementById('wfBar');
        if(wf) {
            wf.innerHTML = steps.map(function(s, i) {
                var cls = i < activeIdx ? 'done' : (i === activeIdx ? 'active' : '');
                return '<span class="wf-s ' + cls + '">' + (i < activeIdx ? '<i class="ri-check-line"></i>' : '') + s + '</span>';
            }).join('');
        }
    }
